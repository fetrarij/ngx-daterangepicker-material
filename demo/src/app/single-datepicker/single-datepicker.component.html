Single datepicker with custom invalid date & custom class on date / with french language
Html code: 
<input type="text"
  ngxDaterangepickerMd
  [isInvalidDate]="isInvalidDate"
  [isCustomDate]="isCustomDate"
  [locale]="locale"
  [(ngModel)]="selected"
  [singleDatePicker]="true"
  [autoApply]="true"/>
Typescript code: 
import dayjs, {{ '{' }} Dayjs } from 'dayjs/esm';
import 'dayjs/esm/locale/fr';
import weekday from 'dayjs/esm/plugin/weekday';
import {{ '{' }} LocaleConfig } from 'ngx-daterangepicker-material';
dayjs.extend(weekday);
dayjs.locale('fr');
...
export class SingleDatepickerComponent {{ '{' }}
  selected = dayjs();
  locale: LocaleConfig = {{ '{' }}
    applyLabel: 'Appliquer',
    customRangeLabel: ' - ',
    daysOfWeek: dayjs.weekdaysMin(),
    monthNames: dayjs.monthsShort(),
    firstDay: dayjs.localeData().firstDayOfWeek()
  };

  isInvalidDate(date: Dayjs): boolean {{ '{' }}
    return date.weekday() === 0;
  }

  isCustomDate(date: Dayjs): 'mycustomdate' | false {{ '{' }}
    return date.weekday() === 0 || date.weekday() === 6 ? 'mycustomdate' : false;
  }
}
